def press(title):
    subprocess.run(['/tmp/aa-live-controls',str(pid),title],check=True)

def state(name):
    requested=time.time()
    time.sleep(.65)
    value=json.loads((out/'live-current.json').read_text())
    assert value['pid']==pid and value['time']>requested, 'passive trace expired or stale; never accept stale assertions'
    (out/(name+'.json')).write_text(json.dumps(value,ensure_ascii=False,indent=2))
    return value
def case(text):
    press('Synthetische Warn-Demo · 6 Fälle');press('UI-Demo · '+text)
    target={'⚠︎ Warnung':'agent-002','‼︎ Kritisch':'agent-003','○ Keine Warnung erkannt':'agent-005'}[text]
    deadline=time.time()+3
    while time.time()<deadline:
        current=json.loads((out/'live-current.json').read_text())
        if current['snapshot']['anchor']==target and current['snapshot']['selection']==target and current['snapshot']['mode']==0:
            time.sleep(.4)
            return
        time.sleep(.05)
    raise AssertionError('menu focus action did not complete before next input')
